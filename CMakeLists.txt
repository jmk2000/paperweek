cmake_minimum_required(VERSION 3.20)
project(paperweek VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PW_BUILD_LVGL "Build the actual LVGL WebAssembly frontend" OFF)
add_library(paperweek_core core/calendar.c)
target_include_directories(paperweek_core PUBLIC core)
target_compile_options(paperweek_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(core_test tests/core_test.c)
  target_link_libraries(core_test PRIVATE paperweek_core)
  add_test(NAME calendar_core COMMAND core_test)
endif()
if(PW_BUILD_LVGL)
  if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Run emcmake cmake for the web LVGL target. Core tests use ordinary CMake.")
  endif()
  include(FetchContent)
  set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
  set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(lvgl GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v9.3.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(lvgl)
  add_executable(paperweek-lvgl platform/web/bridge.c platform/web/display_lvgl.c ui/calendar_ui.c ui/canvas_lvgl.c)
  target_include_directories(paperweek-lvgl PRIVATE core ui)
  target_link_libraries(paperweek-lvgl PRIVATE paperweek_core lvgl::lvgl m)
  set_target_properties(paperweek-lvgl PROPERTIES SUFFIX ".mjs")
  set(PW_EXPORTS "['_pw_init','_pw_configure','_pw_person','_pw_clear_events','_pw_add_event','_pw_select','_pw_clock','_pw_anchor','_pw_is_month','_pw_first','_pw_day_count','_pw_navigate','_pw_help_configure','_pw_press','_pw_visible','_pw_expire','_pw_help_open','_pw_prepare','_pw_draw','_pw_render','_pw_placeholder','_pw_day_rota','_pw_day_items','_pw_day_overflow','_pw_events_count']")
  target_link_options(paperweek-lvgl PRIVATE
    --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createPaperweek
    -sENVIRONMENT=web -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=33554432
    -sMAXIMUM_MEMORY=134217728 -sSTACK_SIZE=262144 -sFILESYSTEM=0
    "-sEXPORTED_FUNCTIONS=${PW_EXPORTS}" "-sEXPORTED_RUNTIME_METHODS=['ccall']")
endif()
